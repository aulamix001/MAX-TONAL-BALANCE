cmake_minimum_required(VERSION 3.22)
project(MaxTonalBalance VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
option(TBC_BUILD_STANDALONE "Build optional standalone applications" OFF)
option(TBC_BUILD_AAX "Build AAX (requires licensed AAX SDK)" OFF)
option(TBC_EXTERNAL_ADAPTERS "Enable future public external adapters" OFF)
if(TBC_BUILD_AAX)
  if(NOT DEFINED TBC_AAX_SDK_PATH)
    message(FATAL_ERROR "Set TBC_AAX_SDK_PATH to your licensed AAX SDK")
  endif()
endif()
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
if(TBC_BUILD_AAX)
  juce_set_aax_sdk_path("${TBC_AAX_SDK_PATH}")
endif()
juce_add_binary_data(tbc_targets HEADER_NAME TbcTargets.h NAMESPACE TbcTargets SOURCES resources/targets/factory.json)
add_library(tbc_core INTERFACE)
target_compile_features(tbc_core INTERFACE cxx_std_20)
set(TBC_FORMATS VST3)
if(TBC_BUILD_STANDALONE)
  list(APPEND TBC_FORMATS Standalone)
endif()
if(TBC_BUILD_AAX)
  list(APPEND TBC_FORMATS AAX)
endif()
function(tbc_plugin target code product track)
  juce_add_plugin(${target}
    COMPANY_NAME "Max Audio" BUNDLE_ID "audio.max.${target}"
    PLUGIN_MANUFACTURER_CODE Mxau PLUGIN_CODE ${code}
    FORMATS ${TBC_FORMATS} PRODUCT_NAME "${product}"
    IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
    COPY_PLUGIN_AFTER_BUILD FALSE)
  target_sources(${target} PRIVATE plugin/Processor.cpp plugin/Editor.cpp)
  target_compile_definitions(${target} PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0 TBC_IS_TRACK=${track})
  target_link_libraries(${target} PRIVATE tbc_core tbc_targets juce::juce_audio_utils
    PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
endfunction()
tbc_plugin(TBC_Track Tbct "TBC Track" 1)
tbc_plugin(TBC_Master Tbcm "TBC Master" 0)
include(CTest)
if(BUILD_TESTING)
  juce_add_console_app(tbc_tests PRODUCT_NAME "TBC state tests")
  target_sources(tbc_tests PRIVATE tests/StateTests.cpp plugin/Processor.cpp plugin/Editor.cpp)
  target_compile_definitions(tbc_tests PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 TBC_IS_TRACK=0)
  target_link_libraries(tbc_tests PRIVATE tbc_targets juce::juce_audio_utils juce::juce_recommended_config_flags)
  add_test(NAME state_and_passthrough COMMAND tbc_tests)
endif()
